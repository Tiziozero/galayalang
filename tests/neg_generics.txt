// Must-fail cases for generics, one program per section.
// Section header:  //### name | text that should appear in the error (optional)
// build.py (`./build.py neg`) splits this file and checks each program fails to compile.

//### conflicting_bindings |
import "core/core.gala";
fn add(a: $T, b: T): T { return a + b; }
fn main() {
    x : i32 = 1;
    y : i64 = 2;
    z := add(x, y);
}

//### float_into_int_binding |
import "core/core.gala";
fn add(a: $T, b: T): T { return a + b; }
fn main() {
    x : i32 = 1;
    z := add(x, 2.5);
}

//### dup_generic_decl | already declared
import "core/core.gala";
fn f(a: $T, b: $T): T { return a; }
fn main() {
    z := f(1, 2);
}

//### use_before_decl | Type "T" doesn't exist.
import "core/core.gala";
fn f(a: T, b: $T): T { return b; }
fn main() {
    z := f(1, 2);
}

//### generic_only_in_return | Can't specify generic in return type
import "core/core.gala";
fn make(a: i32): $T { return a; }
fn main() {
    z := make(1);
}

//### generic_in_vardec | Can't have generics in variable declarations.
import "core/core.gala";
fn f(a: $T) {
    x : $T = a;
}
fn main() {
    f(1);
}
